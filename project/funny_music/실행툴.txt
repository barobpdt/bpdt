parse_colorhunt(db, &s, deleteAll) {
	// https://colorhunt.co/
	if(deleteAll) db.exec('delete from color_map')
	map_idx=1
	node=_node()
	while(s.valid()) {
		s.findPos('<div class="palette">',0,1) not(s.ch()) break;
		ss=s.match('<div','</div>') if(typeof(ss,'bool')) continue;
		ss.findPos('>')
		s.findPos('<span>') not(s.ch()) break;
		num = s.findPos('</span>').trim()
		if(num.find(',')) {
			like_num=num.replace(',','') 
		}
		else {
			like_num=num
		}
		s.findPos('<span class="date">')
		d = s.findPos('</span>').trim()
		while(ss.valid(),n) {
			idx = n+1;
			ss.findPos('background-color: rgb(') not(ss.ch()) break;
			v=ss.findPos(')')
			print("v==$v")
			r=v.findPos(',').trim()
			g=v.findPos(',').trim()	
			b=v.findPos(',').trim()
			c=color(r,g,b)
			color="$c"
			node.with(map_idx, idx, r,g,b,color,like_num,d)
			db.exec("insert into color_map  (map_idx, idx, r,g,b,color,like_num,d) values (#{map_idx}, #{idx}, #{r},#{g},#{b},#{color},#{like_num},#{d})", node)
		}
		map_idx++;
		print("mapIdx==$map_idx")
	}
}
